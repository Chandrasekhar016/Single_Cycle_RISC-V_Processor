//design source of riscv_top module//
`timescale 1ns / 1ps
module riscv_top(
    input  wire        clk,
    input  wire        reset,

    output wire [31:0] debug_pc,
    output wire [31:0] debug_instruction
    );
    // ------------------------------------------------
    // Program Counter and instruction fetch
    // ------------------------------------------------

    reg  [31:0] next_pc;
    wire [31:0] pc;
    wire [31:0] instruction;
    wire [31:0] pc_plus4;

    assign pc_plus4 = pc + 32'd4;

    program_counter u_pc (
        .clk(clk),
        .reset(reset),
        .next_pc(next_pc),
        .pc(pc)
    );

    instruction_memory u_imem (
        .address(pc),
        .instruction(instruction)
    );

    assign debug_pc = pc;
    assign debug_instruction = instruction;

    // ------------------------------------------------
    // Main Control Unit
    // ------------------------------------------------

    wire       reg_write;
    wire       mem_read;
    wire       mem_write;
    wire       mem_to_reg;
    wire       alu_src;
    wire       branch;
    wire       jump;
    wire [1:0] alu_op;
    wire [1:0] result_src;

    main_control u_control (
        .opcode(instruction[6:0]),
        .reg_write(reg_write),
        .mem_read(mem_read),
        .mem_write(mem_write),
        .mem_to_reg(mem_to_reg),
        .alu_src(alu_src),
        .branch(branch),
        .jump(jump),
        .alu_op(alu_op),
        .result_src(result_src)
    );

    // ------------------------------------------------
    // Register File
    // ------------------------------------------------

    wire [31:0] read_data1;
    wire [31:0] read_data2;
    wire [31:0] write_back_data;

    register_file u_regfile (
        .clk(clk),
        .reset(reset),
        .reg_write(reg_write),
        .rs1(instruction[19:15]),
        .rs2(instruction[24:20]),
        .rd(instruction[11:7]),
        .write_data(write_back_data),
        .read_data1(read_data1),
        .read_data2(read_data2)
    );

    // ------------------------------------------------
    // Immediate Generator
    // ------------------------------------------------

    wire [31:0] immediate;

    immediate_generator u_immgen (
        .instruction(instruction),
        .immediate(immediate)
    );

    // ------------------------------------------------
    // ALU Control and ALU
    // ------------------------------------------------

    wire [3:0]  alu_control_code;
    wire [31:0] alu_operand_b;
    wire [31:0] alu_result;
    wire        alu_zero;

    assign alu_operand_b =
        alu_src ? immediate : read_data2;

    alu_control u_alu_control (
        .alu_op(alu_op),
        .funct3(instruction[14:12]),
        .funct7(instruction[31:25]),
        .alu_control(alu_control_code)
    );

    alu u_alu (
        .operand_a(read_data1),
        .operand_b(alu_operand_b),
        .alu_control(alu_control_code),
        .result(alu_result),
        .zero(alu_zero)
    );

    // ------------------------------------------------
    // Data Memory
    // ------------------------------------------------

    wire [31:0] memory_read_data;

    data_memory u_dmem (
        .clk(clk),
        .mem_read(mem_read),
        .mem_write(mem_write),
        .address(alu_result),
        .write_data(read_data2),
        .read_data(memory_read_data)
    );

    // ------------------------------------------------
    // Writeback selection
    // 00 = ALU, 01 = Memory,
    // 10 = PC+4, 11 = Immediate
    // ------------------------------------------------

    reg [31:0] selected_writeback;

    always @(*) begin
        case (result_src)
            2'b00: selected_writeback = alu_result;
            2'b01: selected_writeback = memory_read_data;
            2'b10: selected_writeback = pc_plus4;
            2'b11: selected_writeback = immediate;
            default: selected_writeback = 32'b0;
        endcase
    end

    assign write_back_data = selected_writeback;

    // ------------------------------------------------
    // Next-PC selection
    // ------------------------------------------------

    wire take_branch;
    wire [31:0] branch_target;
    wire [31:0] jump_target;

    assign take_branch = branch & alu_zero;
    assign branch_target = pc + immediate;
    assign jump_target = pc + immediate;

    always @(*) begin
        if (jump)
            next_pc = jump_target;
        else if (take_branch)
            next_pc = branch_target;
        else
            next_pc = pc_plus4;
    end
endmodule

//testbench code//

`timescale 1ns / 1ps
module tb_riscv_top(

    );
    reg clk;
    reg reset;

    wire [31:0] debug_pc;
    wire [31:0] debug_instruction;

    riscv_top dut (
        .clk(clk),
        .reset(reset),
        .debug_pc(debug_pc),
        .debug_instruction(debug_instruction)
    );

    // 10 ns clock period
    initial begin
        clk = 1'b0;
        forever #5 clk = ~clk;
    end

    initial begin
    reset = 1'b1;

    // Reset for two rising edges
    repeat (2) @(posedge clk);
    #1;
    reset = 1'b0;

    // Allow ADDI x1, x0, 10 to execute
    repeat (1) @(posedge clk);
    #1;
    if (dut.u_regfile.registers[1] !== 32'd10)
        $fatal(1, "FAIL: x1 should equal 10");

    // Allow ADDI x2, x0, 20 to execute
    @(posedge clk);
    #1;
    if (dut.u_regfile.registers[2] !== 32'd20)
        $fatal(1, "FAIL: x2 should equal 20");

    // Allow ADD x3, x1, x2 to execute
    @(posedge clk);
    #1;
    if (dut.u_regfile.registers[3] !== 32'd30)
        $fatal(1, "FAIL: x3 should equal 30");

    $display("PASS: Initial processor execution verified.");
    $display("x1 = %0d", dut.u_regfile.registers[1]);
    $display("x2 = %0d", dut.u_regfile.registers[2]);
    $display("x3 = %0d", dut.u_regfile.registers[3]);

    $finish;
end
endmodule
