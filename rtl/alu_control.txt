//design source of alu_control//

`timescale 1ns / 1ps
module alu_control(
    input  wire [1:0] alu_op,
    input  wire [2:0] funct3,
    input  wire [6:0] funct7,

    output reg  [3:0] alu_control
    );
    always @(*) begin

        case (alu_op)

            // Address calculation and immediate arithmetic
            2'b00: begin
                alu_control = 4'b0000; // ADD
            end

            // BEQ: compare operands using subtraction
            2'b01: begin
                alu_control = 4'b0001; // SUB
            end

            // R-type: decode funct3 and funct7
            2'b10: begin
                case (funct3)

                    3'b000: begin
                        if (funct7 == 7'b0100000)
                            alu_control = 4'b0001; // SUB
                        else if (funct7 == 7'b0000000)
                            alu_control = 4'b0000; // ADD
                        else
                            alu_control = 4'b1111; // Unsupported
                    end

                    3'b111: begin
                        if (funct7 == 7'b0000000)
                            alu_control = 4'b0010; // AND
                        else
                            alu_control = 4'b1111;
                    end

                    3'b110: begin
                        if (funct7 == 7'b0000000)
                            alu_control = 4'b0011; // OR
                        else
                            alu_control = 4'b1111;
                    end

                    3'b100: begin
                        if (funct7 == 7'b0000000)
                            alu_control = 4'b0100; // XOR
                        else
                            alu_control = 4'b1111;
                    end

                    3'b010: begin
                        if (funct7 == 7'b0000000)
                            alu_control = 4'b0101; // SLT
                        else
                            alu_control = 4'b1111;
                    end

                    default: alu_control = 4'b1111;

                endcase
            end

            default: alu_control = 4'b1111;

        endcase
    end
endmodule

//testbench code//

`timescale 1ns / 1ps
module tb_alu_control(

    );
     reg [1:0] alu_op;
    reg [2:0] funct3;
    reg [6:0] funct7;

    wire [3:0] alu_control;

    alu_control dut (
        .alu_op(alu_op),
        .funct3(funct3),
        .funct7(funct7),
        .alu_control(alu_control)
    );

    initial begin

        // ADDI, LW, SW: address/arithmetic addition
        alu_op = 2'b00;
        funct3 = 3'b000;
        funct7 = 7'b0000000;
        #10;
        if (alu_control !== 4'b0000)
            $fatal(1, "FAIL: ALUOp ADD");

        // BEQ: subtraction
        alu_op = 2'b01;
        #10;
        if (alu_control !== 4'b0001)
            $fatal(1, "FAIL: ALUOp BEQ");

        // R-type ADD
        alu_op = 2'b10;
        funct3 = 3'b000;
        funct7 = 7'b0000000;
        #10;
        if (alu_control !== 4'b0000)
            $fatal(1, "FAIL: R-type ADD");

        // R-type SUB
        funct7 = 7'b0100000;
        #10;
        if (alu_control !== 4'b0001)
            $fatal(1, "FAIL: R-type SUB");

        // R-type AND
        funct3 = 3'b111;
        funct7 = 7'b0000000;
        #10;
        if (alu_control !== 4'b0010)
            $fatal(1, "FAIL: R-type AND");

        // R-type OR
        funct3 = 3'b110;
        #10;
        if (alu_control !== 4'b0011)
            $fatal(1, "FAIL: R-type OR");

        // R-type XOR
        funct3 = 3'b100;
        #10;
        if (alu_control !== 4'b0100)
            $fatal(1, "FAIL: R-type XOR");

        // R-type SLT
        funct3 = 3'b010;
        #10;
        if (alu_control !== 4'b0101)
            $fatal(1, "FAIL: R-type SLT");

        // Unsupported R-type combination
        funct3 = 3'b001;
        #10;
        if (alu_control !== 4'b1111)
            $fatal(1, "FAIL: Unsupported operation");

        $display("PASS: All ALU control tests completed.");
        $finish;

    end

endmodule
